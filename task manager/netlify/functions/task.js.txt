const { getStore } = require('@netlify/blobs');

exports.handler = async (event) => {
  const store = getStore('tasks');
  const method = event.httpMethod;

  try {
    if (method === 'GET') {
      const { blobs } = await store.list();
      const tasks = await Promise.all(
        blobs.map(async (b) => await store.get(b.key, { type: 'json' }))
      );
      tasks.sort((a, b) => b.created_at - a.created_at);
      return json(200, tasks);
    }

    if (method === 'POST') {
      const { title, description } = JSON.parse(event.body);
      if (!title) return json(400, { error: 'Title required' });

      const id = Date.now().toString() + Math.random().toString(36).slice(2, 8);
      const task = {
        id,
        title,
        description: description || '',
        completed: false,
        created_at: Date.now()
      };

      await store.setJSON(id, task);
      return json(201, task);
    }

    if (method === 'PUT') {
      const { id, completed } = JSON.parse(event.body);
      const task = await store.get(id, { type: 'json' });
      if (!task) return json(404, { error: 'Task not found' });

      task.completed = completed;
      await store.setJSON(id, task);
      return json(200, task);
    }

    if (method === 'DELETE') {
      const id = event.queryStringParameters?.id;
      if (!id) return json(400, { error: 'ID required' });

      await store.delete(id);
      return json(200, { success: true });
    }

    return json(405, { error: 'Method not allowed' });
  } catch (err) {
    console.error('Error:', err);
    return json(500, { error: err.message });
  }
};

function json(statusCode, body) {
  return {
    statusCode,
    headers: {
      'Content-Type': 'application/json',
      'Access-Control-Allow-Origin': '*'
    },
    body: JSON.stringify(body)
  };
}
